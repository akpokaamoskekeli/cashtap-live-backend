require('dotenv').config();
const express = require('express');
const axios = require('axios');
const cors = require('cors');
const app = express();
app.use(cors());
app.use(express.json());

const PAYSTACK_SECRET = process.env.PAYSTACK_SECRET_KEY; // sk_live_...
const PAYSTACK_PUBLIC = process.env.PAYSTACK_PUBLIC_KEY; // pk_live_73c5...

if(!PAYSTACK_SECRET){
  console.warn('WARNING: PAYSTACK_SECRET_KEY not set in .env');
}

function getBankCode(network){
  const n = (network||'').toUpperCase();
  if(n.includes('MTN')) return 'MTN';
  if(n.includes('VOD')) return 'VOD';
  if(n.includes('AIRTEL') || n.includes('TIGO') || n.includes('ATL')) return 'ATL';
  return 'MTN';
}

// Health check
app.get('/', (req,res)=> res.json({status:'CashTap LIVE backend running', live: !!PAYSTACK_SECRET}));

// Create MoMo recipient
async function createRecipient(phone, network){
  // Paystack expects 10 digits without spaces: 0548889608
  const cleanPhone = phone.replace(/\s+/g,'').replace('+233','0');
  const bank_code = getBankCode(network);
  const resp = await axios.post('https://api.paystack.co/transferrecipient',{
    type: 'mobile_money',
    name: 'CashTap Winner',
    account_number: cleanPhone,
    bank_code: bank_code,
    currency: 'GHS'
  },{
    headers: { Authorization: `Bearer ${PAYSTACK_SECRET}` }
  });
  return resp.data.data; // {recipient_code, ...}
}

// Real withdrawal endpoint - THIS SENDS REAL MONEY
app.post('/api/withdraw', async (req,res)=>{
  const { phone, amount, network, userId } = req.body;
  if(!phone || !amount) return res.status(400).json({error:'phone and amount required'});
  if(amount < 1) return res.status(400).json({error:'Min 1 GHS'});
  if(amount > 500) return res.status(400).json({error:'Daily limit 500 GHS'});

  try{
    const recipient = await createRecipient(phone, network);
    const transfer = await axios.post('https://api.paystack.co/transfer',{
      source: 'balance',
      amount: Math.round(amount*100), // pesewas
      recipient: recipient.recipient_code,
      reason: `CashTap winnings - ${userId||'player'} - ${phone}`,
    },{
      headers: { Authorization: `Bearer ${PAYSTACK_SECRET}` }
    });
    // TODO: Save to DB: userId, phone, amount, reference, status
    res.json({
      success: true,
      message: `GHS ${amount} sent to ${phone}`,
      reference: transfer.data.data.reference,
      transfer_code: transfer.data.data.transfer_code,
      recipient: recipient.recipient_code
    });
  }catch(err){
    console.error(err.response?.data || err.message);
    res.status(400).json({error: err.response?.data?.message || err.message, details: err.response?.data});
  }
});

// Verify deposit (called after Paystack inline success)
app.post('/api/verify-deposit', async (req,res)=>{
  const { reference } = req.body;
  try{
    const v = await axios.get(`https://api.paystack.co/transaction/verify/${reference}`,{
      headers:{Authorization:`Bearer ${PAYSTACK_SECRET}`}
    });
    // TODO: credit user balance in DB
    res.json({success:true, data: v.data.data});
  }catch(e){
    res.status(400).json({error:e.response?.data});
  }
});

const PORT = process.env.PORT || 10000;
app.listen(PORT, ()=> console.log(`CashTap LIVE backend on ${PORT}`));
